## 3. 复跑核验

~~~bash
python3 - <<'PY'
import json, subprocess, os, hashlib
d = json.load(open('sources/SOURCE_MANIFEST.json', encoding='utf-8'))
snap = next(sr for sr in d['snapshot_roots'] if 'AI对话录' in sr['root'])
tracked = subprocess.run(['git','ls-files'],capture_output=True,text=True).stdout.splitlines()
alive = drift = gone = 0
for f in snap['files']:
    name = os.path.basename(f['path'])
    hits = [t for t in tracked if os.path.basename(t) == name]
    if any(hashlib.sha256(open(h,'rb').read()).hexdigest()==f['sha256'] for h in hits if os.path.exists(h)):
        alive += 1
    elif hits: drift += 1
    else: gone += 1; print('GONE', f['path'])
print('alive %d / drift %d / gone %d' % (alive, drift, gone))
PY
~~~
